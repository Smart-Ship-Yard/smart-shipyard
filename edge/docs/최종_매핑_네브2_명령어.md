# ── 모든 터미널 ── 환경세팅.
cd ~/smart-shipyard/edge/ros2_ws && source install/setup.bash



# ── 배치 ──────────────────────────────────────
#  ① 배의 긴 변과 로봇을 나란히           ← yaw = 0 이 여기서 정해진다
#  ② 카메라(로봇 오른쪽)에서 배까지 0.8~0.9 m
#     배 앞뒤가 다 보이되 화면 가로를 절반 조금 넘게 채우도록
#  ③ 로봇 앞쪽으로 1.5 m 직진할 공간



# ── 터미널 1 ── 배 중심 측정을 로컬라이제이션 키기 전에 해도 상관 없음. 그저 뎁스카메라로 하기에. 
# ⚠️ 켜기 전 로봇을 캘리브레이션 정면 방향으로! 
# 계정 비번 물으면 치기.
# 욜로를 localization.launch.py가 아닌 systemd가 따로 돌려서 git pull했을 때 최신화 된 욜로코드가 자동 적용이 안됨. 
# 따라서 최신화 된 욜로 코드가 있을 시 해당 코드를 적용하기 위한 장치를 localization.launch.py에 해뒀고 그것이 계정 비번을 요구함.
cd ~/smart-shipyard/edge/ros2_ws && source install/setup.bash
ros2 launch ship_ugv_localization localization.launch.py

# ── 터미널 3 — 배 중심 측정 (로봇 움직이지 말 것) (필요한 것: /camera/center_probe  ← systemd YOLO 서비스가 발행. 로컬라이제이션과 무관)
cd ~/smart-shipyard/edge/ros2_ws/src/ship_ugv_navigation
python3 scripts/measure_ship_center.py --thickness 0.105 # 0.105m가 조준점인 정가운데의 폭이다.



# 통과 기준은 이렇습니다:
# 1. 검출 5개 이상
# 2. 거리 퍼짐 5cm 이하
# 3. 배가 화면에서 안 잘림
# 하나라도 걸리면 저장 안 하고 이유를 알려줍니다. 그러면 고쳐서 다시 치시면 됩니다.



# ── 터미널 2 ── 모션컨트롤 실행
cd ~/smart-shipyard/edge/ros2_ws && source install/setup.bash
ros2 launch ship_ugv_motion_control motion_control.launch.py



# ── 터미널 3 — UWB 캘리브레이션 + 1.5m 직진  ⚠️ 로봇이 움직입니다.
ros2 service call /uwb_map_calibration/calibrate std_srvs/srv/Trigger
ros2 topic pub --once /motion/move_distance std_msgs/msg/Float64 "{data: 1.5}"


# ── 터미널 3 — map → base_link가 (1.5, 0) 근처인지 확인해 보세요.
ros2 run tf2_ros tf2_echo map base_link



# ── 터미널 2 — Ctrl+C 후 매핑
# 배 주위는 안 도셔도 됩니다(측량을 안 쓰니까요). 방 전체를 천천히 훑으시면 됩니다. 벽 근처를 천천히~(스캔 정합이 잘 붙게!).
# 주행 중간에 tf2_echo map base_link로 방위각이 0°에 가까워지는지 봐주시면 수렴 확인됩니다.
ros2 launch ship_ugv_localization mapping.launch.py



# ── 터미널 4 — 키보드 조종
cd ~/smart-shipyard/edge/ros2_ws && source install/setup.bash
ros2 run ship_ugv_motion_control keyboard_teleop_node


# 매핑 주행을 좀 하신 뒤에:
ros2 run tf2_ros tf2_echo map base_link
# 로봇이 +x방향을 보고 배와 나란할 때 yaw가 0° 근처면 수렴한 겁니다.



# ── 터미널 3 — ★ 저장보다 먼저 ★
ros2 service call /slam_map_alignment/align std_srvs/srv/Trigger



# ── 터미널 3 — 저장 (저장할 때 맵이름과 버전번호 기입!)
cd ~/smart-shipyard/edge/ros2_ws/src/ship_ugv_navigation

  --ros-args -p save_map_timeout:=30.0



# ── 터미널 2 Ctrl+C (매핑 끄기) 



# ── 터미널 4 Ctrl+C (키보드 조종 끄기)



# ── 터미널 3→ 마무리
python3 scripts/finalize_map.py shipyard_map_<맵이름>_v<버전번호>
# ★ 여기서 배 방향 재기 — 로봇을 배와 나란히 세우고
python3 scripts/measure_ship_yaw.py shipyard_map_<맵이름>_v<버전번호>
python3 scripts/finalize_map.py shipyard_map_<맵이름>_v<버전번호> --ship-yaw <나온값>

# 순찰 반지름을 변경하고 싶을 때 아래 명령어를 치세요. 기본 설정된 순찰반지름을 그대로 쓸거라면 다음 명령어로 넘어가세요.

python3 scripts/publish_ship_pose.py # 로컬라이제이션 키기 전에 배 중심 측정을 했다면 굳이 하지 않아도 되나, 킨 후에 측정했다면 꼭 해주세요!
